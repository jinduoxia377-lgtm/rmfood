## 二、服务器域名配置

在微信小程序后台配置合法域名：

1. 登录 [微信公众平台](https://mp.weixin.qq.com/)
2. 进入「开发」→「开发管理」→「开发设置」
3. 配置服务器域名：
   - **request 合法域名**：`https://api.rmfood.com`
   - **socket 合法域名**：`wss://api.rmfood.com`
   - **uploadFile 合法域名**：`https://api.rmfood.com`
   - **downloadFile 合法域名**：`https://api.rmfood.com`

## 三、微信登录流程配置

### 后端实现

```python
# backend/app/api/wechat_auth.py

from fastapi import APIRouter, HTTPException, Depends
from pydantic import BaseModel
import httpx
import logging

from ..core.config import settings
from ..services.auth import AuthService
from ..models.user_model import User
from ..core.database.session import get_db_session
from sqlalchemy.ext.asyncio import AsyncSession

router = APIRouter(prefix="/wechat", tags=["微信认证"])
logger = logging.getLogger(__name__)


class WechatLoginRequest(BaseModel):
    code: str


class WechatLoginResponse(BaseModel):
    access_token: str
    refresh_token: str
    expires_in: int
    token_type: str = "Bearer"
    user: dict
    is_new_user: bool


@router.post("/login", response_model=WechatLoginResponse)
async def wechat_login(req: WechatLoginRequest, db: AsyncSession = Depends(get_db_session)):
    """
    微信小程序登录
    
    流程：
    1. 前端调用 wx.login() 获取临时 code
    2. 后端用 code 换取 openid 和 session_key
    3. 生成 JWT Token
    4. 返回 Token 给前端
    """
    # Step 1: 调用微信 code2Session 接口
    wx_data = await _call_wx_code2session(req.code)
    
    if "errcode" in wx_data:
        logger.error(f"微信登录失败: {wx_data}")
        raise HTTPException(
            status_code=400,
            detail=f"微信登录失败: {wx_data.get('errmsg', '未知错误')}"
        )
    
    openid = wx_data["openid"]
    session_key = wx_data.get("session_key")
    
    # Step 2: 查询或创建用户
    service = AuthService(db)
    user, is_new = await service.get_or_create_miniapp_user(
        openid=openid,
        session_key=session_key
    )
    
    # Step 3: 生成 JWT Token
    tokens = await service.generate_jwt_tokens(user)
    
    # Step 4: 返回 Token（不返回 session_key）
    return WechatLoginResponse(
        access_token=tokens.access_token,
        refresh_token=tokens.refresh_token,
        expires_in=tokens.expires_in,
        user={
            "id": user.id,
            "username": user.username,
            "email": user.email,
            "avatar_url": user.avatar_url,
        },
        is_new_user=is_new
    )


async def _call_wx_code2session(code: str) -> dict:
    """调用微信 code2Session 接口"""
    async with httpx.AsyncClient(timeout=10.0) as client:
        resp = await client.get(
            "https://api.weixin.qq.com/sns/jscode2session",
            params={
                "appid": settings.wechat.appid,
                "secret": settings.wechat.secret,
                "js_code": code,
                "grant_type": "authorization_code"
            }
        )
        return resp.json()
```

### 前端实现

```typescript
// src/utils/auth.ts

const API_BASE = 'https://api.rmfood.com/api/v1'

/**
 * 微信小程序登录
 */
export async function wechatLogin(): Promise<{
  token: string
  user: any
  isNewUser: boolean
}> {
  return new Promise((resolve, reject) => {
    // Step 1: 调用 wx.login 获取临时 code
    wx.login({
      success: async (loginRes) => {
        if (!loginRes.code) {
          reject(new Error('获取 code 失败'))
          return
        }
        
        try {
          // Step 2: 发送 code 到后端
          const res = await wx.request({
            url: `${API_BASE}/wechat/login`,
            method: 'POST',
            data: { code: loginRes.code },
            header: { 'Content-Type': 'application/json' }
          })
          
          if (res.statusCode === 200 && res.data.access_token) {
            // Step 3: 存储 Token
            wx.setStorageSync('access_token', res.data.access_token)
            wx.setStorageSync('refresh_token', res.data.refresh_token)
            wx.setStorageSync('user_info', res.data.user)
            
            resolve({
              token: res.data.access_token,
              user: res.data.user,
              isNewUser: res.data.is_new_user
            })
          } else {
            reject(new Error(res.data?.detail || '登录失败'))
          }
        } catch (error) {
          reject(error)
        }
      },
      fail: (error) => {
        reject(error)
      }
    })
  })
}

/**
 * 获取当前 Token
 */
export function getToken(): string {
  return wx.getStorageSync('access_token') || ''
}

/**
 * 检查登录状态
 */
export function isLoggedIn(): boolean {
  const token = getToken()
  return !!token
}

/**
 * 登出
 */
export function logout(): void {
  wx.removeStorageSync('access_token')
  wx.removeStorageSync('refresh_token')
  wx.removeStorageSync('user_info')
}
```
