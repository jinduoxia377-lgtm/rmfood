#### 1.1.2 算法公式还原

```
① 每100g 营养含量（加权平均）：
   Nutrient_per_100g = Σ(原料_i的营养值 × 原料_i添加量) / Σ(所有原料添加量)

② 每件含量 = 每100g × 每份重量(g) / 100

③ NRV% = 每100g含量 ÷ NRV参考值 × 100%

④ 能量转换：kJ = 蛋白质×17 + 脂肪×37 + 碳水化合物×17
          kcal = kJ ÷ 4.184

强制标示项目（GB 28050）：
  能量(kJ)、蛋白质(g)、脂肪(g)、碳水化合物(g)、钠(mg)

可选标示项目：
  饱和脂肪(g)、糖(g)、反式脂肪(g)、膳食纤维(g)、维生素/矿物质
```

**实测验证（原料A示例）：**
- 能量=937kJ，蛋白质=16.2g，脂肪=9g，饱和脂肪=3g，碳水=19.3g，糖=3.3g，钠=671mg
- 反推验证：16.2×17 + 9×37 + 19.3×17 = 275.4+333+328.1 = 936.5 ≈ 937 ✓（公式吻合）

### 2.3 核心计算引擎（新建）

```python
# backend/app/services/nutrition/nutrition_label_generator.py (新建)
"""
预包装食品营养成分表生成器
遵循 GB 28050-2011 食品安全国家标准 预包装食品营养标签通则
"""

from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional, Tuple
from datetime import datetime

# ── NRV参考值（GB 28050 表1）────────────────────────────────
NRV_VALUES = {
    "energy": 8400.0,     # kJ
    "protein": 60.0,      # g
    "fat": 60.0,          # g
    "carbohydrate": 300.0,# g
    "sodium": 2000.0,     # mg
    "saturated_fat": 20.0,# g
    "trans_fat": 0.0,     # g
    "sugar": 50.0,        # g (可选)
    "dietary_fiber": 25.0,# g (可选)
    "cholesterol": 300.0, # mg (可选)
    "calcium": 800.0,     # mg (可选)
    "iron": 15.0,         # mg (可选)
    "zinc": 15.0,         # mg (可选)
    "vitamin_a": 800.0,   # μgRE (可选)
    "vitamin_b1": 1.4,    # mg (可选)
    "vitamin_b2": 1.4,    # mg (可选)
    "vitamin_c": 100.0,   # mg (可选)
    "vitamin_d": 5.0,     # μg (可选)
}

# ── 能量计算系数（kJ/g）────────────────────────────────
CALORIC_CONVERSION = {
    "protein": 17.0,
    "fat": 37.0,
    "carbohydrate": 17.0,
    "dietary_fiber": 8.0,
    "sugar": 17.0,
}

# ── 单位映射（mg ↔ g）────────────────────────────────
MG_TO_G = 0.001
G_TO_MG = 1000.0

# ── 强制标示营养素（GB 28050 第4.1条）────────────────
MANDATORY_NUTRIENTS = ["energy", "protein", "fat", "carbohydrate", "sodium"]


class NutritionLabelGenerator:
    """
    营养成分表生成器

    核心算法（与Excel公式一致）：
    1. 按添加量加权计算各营养素的总量
    2. 归一化为 per_100g
    3. 按比例缩放 per_serving
    4. 计算 NRV%
    5. 能量一致性校验
    """

    def __init__(
        self,
        ingredient_db: Dict[str, Dict[str, float]],
        default_serving_size_g: float = 100.0,
    ):
        """
        Args:
            ingredient_db: 营养数据库 {原料名: {nutrient: value_per_100g}}
            default_serving_size_g: 默认份量(g)
        """
        self._db = ingredient_db
        self._default_serving = default_serving_size_g

    # ── 核心计算 ──────────────────────────────────────────

    def calculate_label(
        self,
        ingredients: List[Dict[str, Any]],
        total_weight_g: float,
        serving_size_g: Optional[float] = None,
        servings_per_container: int = 1,
    ) -> Dict[str, Any]:
        """
        生成完整营养成分表数据

        Args:
            ingredients: [{"name": str, "amount_g": float}]
            total_weight_g: 配方总重量(g)
            serving_size_g: 每份重量(g)，默认等于total_weight
            servings_per_container: 每容器份数

        Returns:
            NutritionLabelResult dict
        """
        if total_weight_g <= 0:
            return self._empty_result("总重量为0，无法计算")

        serving = serving_size_g or total_weight_g

        # Step 1: 加权计算各营养素总量
        total_nutrients = self._weighted_totals(ingredients, total_weight_g)

        # Step 2: 归一化到 per_100g
        per_100g = self._normalize_to_100g(total_nutrients, total_weight_g)

        # Step 3: 计算 per_serving
        per_serving = self._calc_per_serving(per_100g, serving)

        # Step 4: 计算 NRV%
        nrv_pct = self._calc_nrv_percentages(per_100g)

        # Step 5: 能量一致性校验（公式反推）
        energy_check = self._validate_energy(per_100g)

        # Step 6: 营养声称判断
        claims = self._detect_claims(per_100g)

        # Step 7: 合规检查
        compliance = self._check_compliance(per_100g, ingredients)

        # Step 8: 格式化输出
        label_items = self._format_label_items(
            per_100g, per_serving, nrv_pct, claims
        )

        return {
            "formula_id": None,  # 由调用方填充
            "formula_name": "",
            "serving_info": {
                "serving_size_g": serving,
                "servings_per_container": servings_per_container,
                "package_size_g": total_weight_g,
            },
            "per_100g": {k: round(v, 2) for k, v in per_100g.items()},
            "per_serving": {k: round(v, 2) for k, v in per_serving.items()},
            "nrv_percentages": {k: round(v, 2) for k, v in nrv_pct.items()},
            "label_items": label_items,
            "energy_check": energy_check,
            "claims": claims,
            "compliance": compliance,
            "warnings": compliance.get("warnings", []),
            "generated_at": datetime.now().isoformat(),
        }

    # ── 私有计算方法 ──────────────────────────────────────

    def _weighted_totals(
        self, ingredients: List[Dict], total_weight: float
    ) -> Dict[str, float]:
        """
        按添加量加权计算各营养素的绝对总量（单位随输入）

        算法：Σ(原料_i的营养值 × 原料_i添加量_g / 100)
        即：把每100g含量乘以实际用量，得到实际含量
        """
        totals: Dict[str, float] = {}
        unknown = []

        for ing in ingredients:
            name = ing.get("name", "")
            amount_g = ing.get("amount_g", 0) or ing.get("weight_g", 0)
            if amount_g <= 0:
                continue

            db_entry = self._db.get(name)
            if not db_entry:
                unknown.append(name)
                continue

            # 关键：将原料营养值(每100g) × 实际克数 / 100 → 得到该原料贡献的绝对量
            for nutrient, value in db_entry.items():
                # value 是 per_100g，乘以 amount_g/100 得到实际含量
                abs_amount = value * amount_g / 100.0
                totals[nutrient] = totals.get(nutrient, 0.0) + abs_amount

        if unknown:
            # 对于未知原料，尝试用默认值估算（仅蛋白质/脂肪/碳水）
            for name in unknown:
                for nut in ["protein", "fat", "carbohydrate"]:
                    totals[nut] = totals.get(nut, 0.0) + 0.0  # 暂不计入

        return totals

    def _normalize_to_100g(
        self, totals: Dict[str, float], total_weight_g: float
    ) -> Dict[str, float]:
        """
        将绝对总量归一化为每100g含量

        公式：per_100g = total × 100 / total_weight_g
        """
        if total_weight_g <= 0:
            return {}
        factor = 100.0 / total_weight_g
        return {k: v * factor for k, v in totals.items()}

    def _calc_per_serving(
        self, per_100g: Dict[str, float], serving_g: float
    ) -> Dict[str, float]:
        """每份含量 = per_100g × serving_g / 100"""
        return {k: v * serving_g / 100.0 for k, v in per_100g.items()}

    def _calc_nrv_percentages(
        self, per_100g: Dict[str, float]
    ) -> Dict[str, float]:
        """NRV% = per_100g含量 ÷ NRV参考值 × 100"""
        result = {}
        for nutrient, value in per_100g.items():
            nrv = NRV_VALUES.get(nutrient)
            if nrv and nrv > 0:
                result[nutrient] = (value / nrv) * 100
            else:
                result[nutrient] = 0.0
        return result

    def _validate_energy(
        self, per_100g: Dict[str, float]
    ) -> Dict[str, float]:
        """
        能量一致性校验（GB 28050 公式法）

        计算公式：
        能量(kJ) = 蛋白质×17 + 脂肪×37 + 碳水化合物×17 + 膳食纤维×8

        用于验证数据库值的准确性
        """
        protein = per_100g.get("protein", 0)
        fat = per_100g.get("fat", 0)
        carb = per_100g.get("carbohydrate", 0)
        fiber = per_100g.get("dietary_fiber", 0)

        calculated_kj = (
            protein * CALORIC_CONVERSION["protein"]
            + fat * CALORIC_CONVERSION["fat"]
            + carb * CALORIC_CONVERSION["carbohydrate"]
            + fiber * CALORIC_CONVERSION["dietary_fiber"]
        )

        listed_kj = per_100g.get("energy", 0.0)

        return {
            "calculated_kj": round(calculated_kj, 1),
            "listed_kj": round(listed_kj, 1),
            "diff_kj": round(abs(calculated_kj - listed_kj), 1),
            "match": abs(calculated_kj - listed_kj) < 10.0,  # 允许±10kJ误差
        }

    def _detect_claims(self, per_100g: Dict[str, float]) -> List[str]:
        """
        检测营养声称（GB 28050 附录A）

        常见声称门槛（每100g或每100ml）：
        - "高蛋白质": protein ≥ 12g/100g (固体) 或 ≥ 6g/100ml (液体)
        - "低脂肪": fat ≤ 3g/100g (固体) 或 ≤ 1.5g/100ml (液体)
        - "高纤维": dietary_fiber ≥ 6g/100g
        - "低钠": sodium ≤ 120mg/100g
        - "无糖": sugar ≤ 0.5g/100g
        - "低饱和脂肪": sat_fat ≤ 1.5g/100g
        """
        claims = []
        protein = per_100g.get("protein", 0)
        fat = per_100g.get("fat", 0)
        fiber = per_100g.get("dietary_fiber", 0)
        sodium = per_100g.get("sodium", 0)
        sugar = per_100g.get("sugar", 0)
        sat_fat = per_100g.get("saturated_fat", 0)

        if protein >= 12:
            claims.append(f"高蛋白质 ({protein}g/100g)")
        if fat <= 3:
            claims.append(f"低脂肪 ({fat}g/100g)")
        if fiber >= 6:
            claims.append(f"高膳食纤维 ({fiber}g/100g)")
        if sodium <= 120:
            claims.append(f"低钠 ({sodium}mg/100g)")
        if sugar <= 0.5:
            claims.append(f"无糖 ({sugar}g/100g)")
        if sat_fat > 0 and sat_fat <= 1.5:
            claims.append(f"低饱和脂肪 ({sat_fat}g/100g)")

        return claims

    def _check_compliance(
        self,
        per_100g: Dict[str, float],
        ingredients: List[Dict],
    ) -> Dict[str, Any]:
        """GB 28050 合规性检查"""
        violations = []
        warnings = []

        # 强制项目完整性检查
        for nut in MANDATORY_NUTRIENTS:
            if nut not in per_100g:
                violations.append({
                    "type": "missing_mandatory",
                    "nutrient": nut,
                    "message": f"缺少强制标示营养素: {nut}",
                })

        # 钠含量警告
        sodium = per_100g.get("sodium", 0)
        if sodium > 600:
            warnings.append({
                "type": "high_sodium",
                "value": sodium,
                "message": f"钠含量较高({sodium}mg/100g)，建议标注'高钠'警示",
            })

        # 能量一致性警告
        energy = per_100g.get("energy", 0)
        if energy > 0:
            expected = (
                per_100g.get("protein", 0) * 17
                + per_100g.get("fat", 0) * 37
                + per_100g.get("carbohydrate", 0) * 17
            )
            if abs(energy - expected) > energy * 0.1:
                warnings.append({
                    "type": "energy_mismatch",
                    "listed": energy,
                    "calculated": round(expected, 1),
                    "message": "能量值与宏量营养素计算值偏差超过10%，请核查",
                })

        return {
            "is_compliant": len(violations) == 0,
            "standard": "GB 28050-2011",
            "violations": violations,
            "warnings": warnings,
        }

    def _format_label_items(
        self,
        per_100g: Dict[str, float],
        per_serving: Dict[str, float],
        nrv_pct: Dict[str, float],
        claims: List[str],
    ) -> List[Dict[str, Any]]:
        """格式化为GB 28050标准标签行格式"""
        # 定义显示顺序和单位
        display_order = [
            ("energy", "kJ", True),
            ("protein", "g", True),
            ("fat", "g", True),
            ("saturated_fat", "g", False),
            ("carbohydrate", "g", True),
            ("sugar", "g", False),
            ("dietary_fiber", "g", False),
            ("sodium", "mg", True),
        ]

        items = []
        for nutrient, unit, is_mandatory in display_order:
            value_100g = per_100g.get(nutrient, 0.0)
            value_serving = per_serving.get(nutrient, 0.0)
            nrv = nrv_pct.get(nutrient, 0.0)

            item = {
                "nutrient": nutrient,
                "unit": unit,
                "is_mandatory": is_mandatory,
                "per_100g": round(value_100g, 2),
                "per_serving": round(value_serving, 2) if value_serving > 0 else None,
                "nrv_percent": round(nrv, 1) if nrv > 0 else None,
            }

            # 添加营养声称标记
            if is_mandatory or value_100g > 0:
                items.append(item)

        return items

    def _empty_result(self, reason: str) -> Dict[str, Any]:
        return {
            "error": reason,
            "per_100g": {},
            "per_serving": {},
            "nrv_percentages": {},
            "label_items": [],
            "energy_check": {},
            "compliance": {"is_compliant": False, "violations": [{"type": "input_error", "message": reason}]},
            "warnings": [reason],
        }


# ── 单例工厂 ──────────────────────────────────────────────

) -> NutritionLabelGenerator:
    global _instance
    if _instance is None or ingredient_db is not None:
```

```python
    返回：
    - per_100g: 每100g营养成分含量
    - per_serving: 每份营养成分含量
    - nrv_percentages: 各营养素NRV%

        get_nutrition_label_generator,
        HERB_NUTRITION_DB,
    )

    formula = await db.get(Formula, formula_id)
    if not formula:
        raise HTTPException(status_code=404, detail="配方不存在")

    # 提取成分数据
    ingredients = []
    for ing in formula.ingredients:
        amount_g = ing.properties.get("amount_g", 0) or ing.properties.get("weight_g", 0)
        if amount_g > 0:
            ingredients.append({"name": ing.name, "amount_g": amount_g})

    total_weight = sum(i["amount_g"] for i in ingredients)

        servings_per_container=servings_per_container,
    )
    result["formula_id"] = formula_id
    result["formula_name"] = formula.name

    return result
```
